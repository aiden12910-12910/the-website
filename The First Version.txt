    <script>
        document.addEventListener('keydown', function(event) {
            // 1. 紧急隐藏：按左上角 Esc 键，0.01 秒内把当前网页彻底变更为 Google Docs
            if (event.key === 'Escape') {
                sessionStorage.setItem('restoreCheckpoint', window.location.href);
                window.location.href = 'https://google.com';
            }
            
            // 2. 悄悄变回：警报解除后，在同一个标签页按下数字 1 左边的 ` 键（波浪号 ~ 键），瞬间回到摸鱼列表
            if (event.key === '`') {
                let savedUrl = sessionStorage.getItem('restoreCheckpoint');
                if (savedUrl) {
                    window.location.href = savedUrl;
                } else {
                    window.location.href = 'https://github.io';
                }
            }
        });
    </script>
